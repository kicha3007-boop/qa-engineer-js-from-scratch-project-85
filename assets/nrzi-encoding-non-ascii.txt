Исходник C++:
#include <fstream>
#include <iostream>
#include <string>
std::string solution(const std::string& s) {
  std::string r;
  for (size_t i = 0; i < s.size(); i++) {
    if (s[i] == '|') continue;
    r += (i > 0 && s[i - 1] == '|') ? '1' : '0';
  }
  return r;
}
int main() {
  std::ifstream in("cases.txt");
  std::string line; int ok = 0, all = 0;
  while (std::getline(in, line)) {
    auto tab = line.find('\t');
    std::string arg = line.substr(0, tab), exp = line.substr(tab + 1);
    std::string got = solution(arg);
    all++;
    if (got == exp) ok++;
    else if (all - ok <= 3) std::cout << "FAIL arg=" << arg << " bytes=" << arg.size() << " expected=" << exp << " got=" << got << "\n";
  }
  std::cout << "C++ (std::string bytes): passed " << ok << " of " << all << "\n";
}
